- Customers who send you a drill sheet to have a ball drilled already have what RollRead needs.

- Coaching sources say oversized holes make you squeeze, and squeezing limits hook.
- Six drills, 10 to 15 minutes a day: fingertip carries, thumb pops, swing sync, cupped-wrist isometrics, carpet spins, rotation-axis control
- Designed to practice a relaxed fingertip hold and a clean, early thumb exit, in the bowler's real fit

**Speaker note:** Lead with fit. Fit is a fact we can show on a bench. The coaching statements are attributed consensus, and we present them that way: "coaches agree", "the trade body puts".               'thumb_fwd', 'thumb_lat', 'middle_fwd', 'middle_lat', 'ring_fwd', 'ring_lat'];

  function setField(key, val) {
    var el = root.querySelector('[data-fs="' + key + '"]');
    if (el && val !== undefined && val !== null) el.value = String(val);
  }

  // Fill the fields from a pasted configurator fit sheet (JSON, sizes and spans in mm,
  // pitches in drill-sheet inches). Non-JSON text is left as typed for us to read.
  if (paste) {
    paste.addEventListener('change', function () {
      var txt = paste.value.trim();
      if (!txt) return;
      try {
        var sheet = JSON.parse(txt);
        keys.forEach(function (k) { setField(k, sheet[k]); });
        if (units) units.value = 'mm';
        if (hand && sheet.hand) {
          var h = String(sheet.hand).toLowerCase();
          hand.value = h.charAt(0) === 'l' ? 'Left' : 'Right';
          hand.dispatchEvent(new Event('change', { bubbles: true }));
        }
      } catch (e) { /* not JSON */ }
    });
  }

  // Keep the Hand variant in step with the Hand property when the product has a Hand option.
  if (hand) {
    hand.addEventListener('change', function () {
      var form = root.closest('form');
      if (!form || !hand.value) return;
      var radio = form.querySelector('input[name="Hand"][value="' + hand.value + '"]');
      if (radio) {
        if (!radio.checked) { radio.checked = true; radio.dispatchEvent(new Event('change', { bubbles: true })); }
        return;
      }
      var sel = form.querySelector('select[name="options[Hand]"]');
      if (sel && sel.value !== hand.value) {
        sel.value = hand.value;
        sel.dispatchEvent(new Event('change', { bubbles: true }));
      }
    });
  }
})();
</script>
{%- endif -%}

Notes for the theme developer:

- Line-item property names match the fit-sheet field map in Section C, so an order export maps onto the build pipeline without renaming.
- `_Referral` starts with an underscore, so Shopify hides it from the customer in the cart and checkout while keeping it on the order for attribution.
- The `required` attributes block add-to-cart until the fields are filled. Themes that add to cart over AJAX still submit the properties; themes that bypass form validation should call `form.reportValidity()` before posting.
- The variant-sync script looks for the two common variant-picker markups (radio inputs named `Hand`, or a select named `options[Hand]`). If the theme uses another markup, adjust the two selectors.
- Style the `rollread-fit__*` classes in the theme's CSS. The snippet carries no styles of its own.
